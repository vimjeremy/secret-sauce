* Straight.el
#+begin_src emacs-lisp
  (defvar bootstrap-version)
  (let ((bootstrap-file
         (expand-file-name
          "straight/repos/straight.el/bootstrap.el"
          (or (bound-and-true-p straight-base-dir)
              user-emacs-directory)))
        (bootstrap-version 7))
    (unless (file-exists-p bootstrap-file)
      (with-current-buffer
          (url-retrieve-synchronously
           "https://raw.githubusercontent.com/radian-software/straight.el/develop/install.el"
           'silent 'inhibit-cookies)
        (goto-char (point-max))
        (eval-print-last-sexp)))
    (load bootstrap-file nil 'nomessage))

  (straight-use-package 'use-package)
  (setq straight-use-package-by-default t)
  (use-package general)
#+end_src

  (use-package emacs
    :init
    (setq use-short-answers t
          custom-file "~/.config/emacs/custom.el"
  	      scroll-conservatively 101
          auto-save-list-file-prefix "~/.Trash/"
          backup-directory-alist            '((".*" . "~/.Trash"))
          auto-save-file-name-transforms '((".*" "~/.Trash/" t))
          default-frame-alist '((font . "Iosevka 22"))
          inhibit-startup-screen t)
    (when (file-exists-p custom-file) (load custom-file))
    :config
    (setq-default truncate-lines t
  		            display-line-numbers-width 3
  		            indent-tabs-mode nil
  		            tab-width 2)
    (tool-bar-mode -1)
    (menu-bar-mode -1)
    (scroll-bar-mode -1)
    (global-display-line-numbers-mode 1)
    (fringe-mode '(8 . 8))
    :custom
    (tab-always-indent 'complete)
    (text-mode-ispell-word-completion nil)
    (read-extended-command-predicate #'command-completion-default-include-p)
    :general-config
    ("<escape>" 'keyboard-escape-quit))

* Dired
#+begin_src emacs-lisp
  (defun wdired-action(action)
    (interactive)
    (wdired-change-to-wdired-mode)
    (call-interactively action))

  (use-package dired
    :after evil
    :straight nil
    :custom (dired-listing-switches "--group-directories-first -lAhX")
    :general
    (:keymaps 'dired-mode-map
              :states 'normal
              "I" (lambda () (interactive) (wdired-action #'evil-insert-line))
              "A" (lambda () (interactive) (wdired-action #'evil-append-line))
              "a" (lambda () (interactive) (wdired-action #'evil-append))
              "i" (lambda () (interactive) (wdired-action #'evil-insert))))
#+end_src
